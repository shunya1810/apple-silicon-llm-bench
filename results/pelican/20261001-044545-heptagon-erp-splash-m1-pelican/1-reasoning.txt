```
def apply_wall_collision(ball: Ball, edges: List[Tuple[...]], omega: float, position_only: bool) -> None:
    r = ball.radius
    x = ball.x
    y = ball.y
    vx = ball.vx
    vy = ball.vy
    w = ball.angular_vel
    inv_I = 1.0 / ball.inertia
    for (x0, y0, ax, ay, a_len2, nx, ny) in edges:
        d = (x - x0) * nx + (y - y0) * ny
        if d < r:
            # project
            if a_len2 > 1e-12:
                t = ((x - x0) * ax + (y - y0) * ay) / a_len2
            else:
                t = 0.0
            if t < 0.0: t = 0.0
            elif t > 1.0: t = 1.0
            qx = x0 + ax * t
            qy = y0 + ay * t
            push = r - d
            x += nx * push
            y += ny * push
            if not position_only:
                # recompute contact q with updated position
                if a_len2 > 1e-12:
                    t = ((x - x0) * ax + (y - y0) * ay) / a_len2
                    if t < 0.0: t = 0.0
                    elif t > 1.0: t = 1.0
                    qx = x0 + ax * t
                    qy = y0 + ay * t
                wall_vx = -omega * (qy - CENTER_Y)
                wall_vy = omega * (qx - CENTER_X)
                rvx = vx - wall_vx
                rvy = vy - wall_vy
                vn = rvx * nx + rvy * ny
                if vn < 0.0:
                    jn = (1.0 + WALL_RESTITUTION) * (-vn)
                    tx = -ny
                    ty = nx
                    vt = rvx * tx + rvy * ty
                    s = vt - w * r
                    if abs(s) > 1e-8:
                        jt_un = -s / (1.0 + (r * r) * inv_I)
                        jt_max = WALL_FRICTION * jn
                        if jt_un > jt_max:
                            jt = jt_max
                        elif jt_un < -jt_max:
                            jt = -jt_max
                        else:
                            jt = jt_un
                    else:
                        jt = 0.0
                    vx += jn * nx + jt * tx
                    vy += jn * ny + jt * ty
                    w += -jt * r * inv_I
    ball.x = x
    ball.y = y
    ball.vx = vx
    ball.vy = vy
    ball.angular_vel = w
```

```
def resolve_ball_collisions(balls: List[Ball]) -> None:
    n = len(balls)
    min_dist = 2.0 * balls[0].radius if n else 0.0
    min_dist2 = min_dist * min_dist
    slop = 0.02
    for i in range(n):
        bi = balls[i]
        for j in range(i+1,n):
            bj = balls[j]
            dx = bj.x - bi.x
            dy = bj.y - bi.y
            dist2 = dx*dx + dy*dy
            if dist2 >= min_dist2:
                continue
            if dist2 > 1e-12:
                dist = math.sqrt(dist2)
                nx = dx / dist
                ny = dy / dist
            else:
                dist = 1e-9
                angle = float(i * 7 + j * 13) * 0.37
                nx = math.cos(angle)
                ny = math.sin(angle)
            penetration = min_dist - dist
            correction = max(penetration - slop, 0.0) * 0.8
            if correction > 0.0:
                cx = 0.5 * correction * nx
                cy = 0.5 * correction * ny
                bi.x -= cx; bi.y -= cy
                bj.x += cx; bj.y += cy
            r = balls[0].radius
            # contact velocities
            vi_cx = bi.vx - bi.angular_vel * r * ny
            vi_cy = bi.vy + bi.angular_vel * r * nx
            vj_cx = bj.vx + bj.angular_vel * r * ny
            vj_cy = bj.vy - bj.angular_vel * r * nx
            rvx = vj_cx - vi_cx
            rvy = vj_cy - vi_cy
            vn = rvx * nx + rvy * ny
            if vn < 0.0:
                inv_m = 1.0 / bi.mass
                inv_i = 1.0 / bi.inertia
                jn = -(1.0 + BALL_RESTITUTION) * vn / (inv_m + inv_m)
                tx = -ny
                ty = nx
                vt = rvx * tx + rvy * ty
                denom_t = inv_m + inv_m + (r * r) * inv_i + (r * r) * inv_i
                jt_un = -vt / denom_t
                jt_max = BALL_FRICTION * jn
                if jt_un > jt_max:
                    jt = jt_max
                elif jt_un < -jt_max:
                    jt = -jt_max
                else:
                    jt = jt_un
                ix = -jn * nx - jt * tx
                iy = -jn * ny - jt * ty
                jx = jn * nx + jt * tx
                jy = jn * ny + jt * ty
                bi.vx += ix
                bi.vy += iy
                bj.vx += jx
                bj.vy += jy
                dw = -jt * r
                bi.angular_vel += dw / bi.inertia
                bj.angular_vel += dw / bj.inertia
```
