```
def closest_on_segment(px,py, ax,ay, bx,by):
    abx=bx-ax; aby=by-ay
    apx=px-ax; apy=py-ay
    denom=abx*abx+aby*aby
    t=(apx*abx+apy*aby)/denom
    if t<0: t=0
    elif t>1: t=1
    return ax+abx*t, ay+aby*t
```

```
def _heptagon_state(self, theta: float) -> Tuple[List[Tuple[float,float]], List[Tuple[float,float]]]:
    verts=[]
    for i in range(N_SIDES):
       a=theta + i*(2*math.pi/N_SIDES)
       verts.append((CENTER_X + R_HEPT*math.cos(a), CENTER_Y + R_HEPT*math.sin(a)))
    norms=[]
    for i in range(N_SIDES):
       ax,ay=verts[i]; bx,by=verts[(i+1)%N_SIDES]
       mx=(ax+bx)*0.5; my=(ay+by)*0.5
       nx=CENTER_X-mx; ny=CENTER_Y-my
       nl=math.hypot(nx,ny)
       if nl < 1e-12:
          nx,ny=1.0,0.0
       else:
          nx/=nl; ny/=nl
       norms.append((nx,ny))
    return verts, norms
```

```
def closest_on_segment(px: float, py: float, ax: float, ay: float, bx: float, by: float) -> Tuple[float,float,float]:
    abx=bx-ax; aby=by-ay
    denom=abx*abx+aby*aby
    if denom < 1e-12:
       return ax, ay, 0.0
    t=((px-ax)*abx + (py-ay)*aby)/denom
    if t < 0.0: t=0.0
    elif t > 1.0: t=1.0
    return ax+abx*t, ay+aby*t, t
```
