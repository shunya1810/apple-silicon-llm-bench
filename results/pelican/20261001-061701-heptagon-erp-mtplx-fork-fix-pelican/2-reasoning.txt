def polygon_sdf(q: Vector) -> Tuple[float, Vector, Vector, bool]:
    inside = point_inside(q)
    best_p = (0.0,0.0)
    best_d2 = float('inf')
    for a,b in HEPTA_EDGES:
       p = closest_point_on_segment(q,a,b)
       dx=q[0]-p[0]; dy=q[1]-p[1]
       d2=dx*dx+dy*dy
       if d2 < best_d2:
          best_d2=d2; best_p=p
    dist=math.sqrt(best_d2)
    if dist < 1e-9:
       L=math.hypot(best_p[0], best_p[1])
       if L < 1e-9:
          n=(1.0,0.0)
       else:
          n=(-best_p[0]/L, -best_p[1]/L)
       s=0.0
    elif inside:
       n=((q[0]-best_p[0])/dist, (q[1]-best_p[1])/dist)
       s=dist
    else:
       n=((best_p[0]-q[0])/dist, (best_p[1]-q[1])/dist)
       s=-dist
    return s,n,best_p,inside
