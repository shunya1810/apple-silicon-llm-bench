   ```rust
   fn parse_iso8601_duration(s: &str) -> Result<f64, String> {
       if !s.starts_with('P') {
           return Err("Duration must start with 'P'".to_string());
       }
       let mut s = s.trim_start_matches('P');
       let mut total_seconds = 0.0;

       // Find 'T' to split date and time parts
       let time_start = s.find('T').unwrap_or(s.len());
       let date_part = &s[..time_start];
       let time_part = &s[time_start + 1..];

       // Parse date part: Y, M, D
       total_seconds += parse_component(date_part, 'Y') * 365.2425 * 24.0 * 3600.0;
       total_seconds += parse_component(date_part, 'M') * (365.2425 / 12.0) * 24.0 * 3600.0;
       total_seconds += parse_component(date_part, 'D') * 24.0 * 3600.0;

       // Parse time part: H, M, S
       total_seconds += parse_component(time_part, 'H') * 3600.0;
       total_seconds += parse_component(time_part, 'M') * 60.0;
       total_seconds += parse_component(time_part, 'S');

       Ok(total_seconds)
   }

   fn parse_component(part: &str, unit: char) -> f64 {
       match part.find(unit) {
           None => 0.0,
           Some(pos) => {
               let num_str = &part[..pos];
               num_str.parse::<f64>().unwrap_or(0.0)
           }
       }
   }
   ```
   Wait, `parse_component` logic is a bit flawed if multiple units exist. For example, `P3D4H` -> `date_part` is `3D`, `time_part` is `4H`. `parse_component` for `'D'` in `3D` works. But what if `P3D4H5M`? `date_part` = `3D`, `time_part` = `4H5M`. `parse_component("4H5M", 'H')` finds `H` at pos 1, takes `"4"`, parses to 4.
