   pub fn parse_iso_duration_to_seconds(s: &str) -> Result<i64, String> {
       if !s.starts_with('P') {
           return Err("Duration must start with 'P'".to_string());
       }

       let mut chars = s.chars().peekable();
       chars.next(); // skip 'P'

       let mut days: i64 = 0;
       let mut hours: i64 = 0;
       let mut minutes: i64 = 0;
       let mut seconds: i64 = 0;
       let mut time_part = false;

       loop {
           match chars.next() {
               None => break,
               c if c.is_ascii_digit() => {
                   let mut num: i64 = 0;
                   while let Some(ch) = chars.peek() {
                       if ch.is_ascii_digit() {
                           num = num * 10 + (ch as i64 - '0' as i64);
                           chars.next();
                       } else {
                           break;
                       }
                   }
                   match chars.next() {
                       Some('Y') => return Err("Years cannot be precisely converted to seconds".to_string()),
                       Some('M') => {
                           if !time_part {
                               return Err("Months cannot be precisely converted to seconds".to_string());
                           }

                       Some('S') => seconds += num,
                       other => return Err(format!("Invalid unit '{}'. Expected Y, M, D, T, H, or S.", other)),
                   }
               }
               _ => return Err("Expected a digit".to_string()),
           }
       }
